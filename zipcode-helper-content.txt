        <h2>Instructions</h2>
        <ol>
            <li>Use the "Load Current Data" button to view the current zipcode data</li>
            <li>Edit the data in the textarea as needed</li>
            <li>Click "Save Zipcode Data" to update the system</li>
        </ol>
        <p>The data should be in GeoJSON format with properties including "zipcode", "name", and "isValid" flags.</p>

    <script>
        document.addEventListener('DOMContentLoaded', function() {
            const currentDataElement = document.getElementById('currentData');
            const zipcodeDataTextarea = document.getElementById('zipcodeData');
            const statusElement = document.getElementById('status');
            const loadButton = document.getElementById('loadCurrentData');
            const saveButton = document.getElementById('saveZipcodes');
            
            // Function to load current data
            function loadCurrentData() {
                statusElement.innerHTML = '';
                statusElement.className = '';
                
                fetch('/api/zipcodes')
                    .then(response => response.json())
                    .then(data => {
                        const formattedData = JSON.stringify(data, null, 2);
                        currentDataElement.textContent = formattedData;
                        zipcodeDataTextarea.value = formattedData;
                    })
                    .catch(error => {
                        console.error('Error fetching zipcode data:', error);
                        currentDataElement.textContent = 'Error loading data: ' + error.message;
                        statusElement.innerHTML = 'Failed to load current data';
                        statusElement.className = 'error';
                    });
            }
            
            // Function to save zipcode data
            function saveZipcodeData() {
                try {
                    const data = JSON.parse(zipcodeDataTextarea.value);
                    
                    statusElement.innerHTML = 'Saving data...';
                    statusElement.className = '';
                    
                    fetch('/api/save-zipcode-data', {
                        method: 'POST',
                        headers: {
                            'Content-Type': 'application/json'
                        },
                        body: JSON.stringify(data)
                    })
                    .then(response => {
                        if (!response.ok) {
                            throw new Error('Server returned status: ' + response.status);
                        }
                        return response.json();
                    })
                    .then(result => {
                        statusElement.innerHTML = 'Data saved successfully!';
                        statusElement.className = 'success';
                        loadCurrentData(); // Reload the current data
                    })
                    .catch(error => {
                        console.error('Error saving zipcode data:', error);
                        statusElement.innerHTML = 'Error saving data: ' + error.message;
                        statusElement.className = 'error';
                    });
                } catch (e) {
                    statusElement.innerHTML = 'Invalid JSON format: ' + e.message;
                    statusElement.className = 'error';
                }
            }
            
            // Set up event listeners
            loadButton.addEventListener('click', loadCurrentData);
            saveButton.addEventListener('click', saveZipcodeData);
            
            // Load data on page load
            loadCurrentData();
        });
    </script>
